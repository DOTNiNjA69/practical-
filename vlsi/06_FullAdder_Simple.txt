EXPERIMENT 6: FULL ADDER USING TWO HALF ADDERS
================================================

HALF ADDER DESIGN
-----------------
module half_adder(input a,b, output sum,carry);
assign sum = a ^ b;
assign carry = a & b;
endmodule

FULL ADDER DESIGN
-----------------
module full_adder(input a,b,cin, output sum,cout);

wire s1,c1,c2;

half_adder h1(a,b,s1,c1);
half_adder h2(s1,cin,sum,c2);

assign cout = c1 | c2;
endmodule

TESTBENCH
----------
module full_adder_tb;
reg a,b,cin;
wire sum,cout;

full_adder uut(a,b,cin,sum,cout);

initial begin
    $monitor("A=%b B=%b Cin=%b Sum=%b Carry=%b",a,b,cin,sum,cout);

    a=0; b=0; cin=0; #10;
    a=0; b=0; cin=1; #10;
    a=0; b=1; cin=0; #10;
    a=0; b=1; cin=1; #10;
    a=1; b=0; cin=0; #10;
    a=1; b=0; cin=1; #10;
    a=1; b=1; cin=0; #10;
    a=1; b=1; cin=1; #10;

    $finish;
end
endmodule

EASY TO REMEMBER
----------------
First half adder adds A and B.
Second half adder adds first sum and Cin.
OR the two carry outputs to get final carry.
