// EXPERIMENT 1: SISO SHIFT REGISTER

module siso_top(
    input clk_100,
    input rst,
    input serial_in,
    output serial_out
);

reg [25:0] cnt;
reg slow_clk;
reg [3:0] data;

// Clock Divider: 100 MHz to 1 Hz
always @(posedge clk_100 or posedge rst) begin
    if(rst) begin
        cnt <= 0;
        slow_clk <= 0;
    end
    else if(cnt == 49999999) begin
        cnt <= 0;
        slow_clk <= ~slow_clk;
    end
    else
        cnt <= cnt + 1;
end

// SISO Shift Register
always @(posedge slow_clk or posedge rst) begin
    if(rst)
        data <= 4'b0000;
    else
        data <= {data[2:0], serial_in};
end

assign serial_out = data[3];

endmodule

`timescale 1ns / 1ps

module siso_tb;

reg clk_100, rst, serial_in;
wire serial_out;

siso_top uut(clk_100, rst, serial_in, serial_out);

// 100 MHz Clock Generation
always #5 clk_100 = ~clk_100;

initial begin
    clk_100 = 0;
    rst = 1;
    serial_in = 0;

    #20;
    rst = 0;

    // Apply serial inputs
    serial_in = 1; #1000000000;
    serial_in = 0; #1000000000;
    serial_in = 1; #1000000000;
    serial_in = 1; #1000000000;
    serial_in = 0; #1000000000;

    $finish;
end

initial begin
    $monitor("Time=%0t Input=%b Output=%b",
             $time, serial_in, serial_out);
end

endmodule
