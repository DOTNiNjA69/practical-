#include <reg51.h>
sbit DIG1 = P3^4;
sbit DIG2 = P3^5;
unsigned char code seg[16] = {
    0x3F, 0x06, 0x5B, 0x4F,
    0x66, 0x6D, 0x7D, 0x07,
    0x7F, 0x6F, 0x77, 0x7C,
    0x39, 0x5E, 0x79, 0x71
};
void short_delay(void)
{
    unsigned int i;
    for(i = 0; i < 700; i++);
}
void refresh_display(unsigned char value)
{
    unsigned char high, low;
    high = value >> 4;
    low  = value & 0x0F;
    DIG1 = 1; DIG2 = 1;       /* Both digits OFF */
    P2 = seg[high];
    DIG1 = 0;                 /* Left digit ON */
    short_delay();
    DIG1 = 1;
    P2 = seg[low];
    DIG2 = 0;                 /* Right digit ON */
    short_delay();
    DIG2 = 1;
}
void main(void)
{
    unsigned int n, r;
    DIG1 = 1;
    DIG2 = 1;
    while(1)
    {
        for(n = 0; n < 256; n++)
        {
            for(r = 0; r < 100; r++)
                refresh_display((unsigned char)n);
        }
    }
}
